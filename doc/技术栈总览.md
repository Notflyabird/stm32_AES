## 1. 启动流程实现

**文件**: `Core/Src/main.c`

### 主函数流程

```c
int main(void)
{
    HAL_Init();
    SystemClock_Config();                     // HSI 8MHz → PLL x8 → 64MHz
    MX_GPIO_Init();
    MX_DMA_Init();
    MX_USART1_UART_Init();
    MX_CAN_Init();
    uds_init();                               // UDS 协议栈初始化

    // 第一阶段: 检查 RAM 回退标志
    if (check_app_set_flag() == 1)
    {
        // APP 要求回退到 Bootloader
        uart_print_hex(BACK_TO_BOOT_FLAG_ADDR, 16, "stay FBL");
        *(uint32_t *)BACK_TO_BOOT_FLAG_ADDR = 0x00000000;  // 清除标志
    }
    else
    {
        // 第二阶段: CAN 请求检测
        if (can_start_check() == 0)           // 超时未收到 CAN 请求
        {
            uart_print_hex(key, 16, "CAN Check Failed: ");
            // 第三阶段: 检查 APP 有效标志
            if (*(uint32_t *)APP_VALID_FLAG_ADDR == 0x5A5A5A5A)
            {
                Boot_JumpToApplication();     // 跳转到 APP
            }
        }
        else                                  // 收到 CAN 请求，停留在 FBL
        {
            uart_print_hex(key, 16, "CAN Check Passed: ");
        }
    }

    MX_IWDG_Init();                           // 看门狗在判决之后初始化

    // 主循环
    while (1)
    {
        static uint32_t cnt_1ms, led_cnt;
        if (HAL_GetTick() >= cnt_1ms + 1)
        {
            cnt_1ms = HAL_GetTick();
            uds_1ms_task();                   // UDS 定时任务 (1ms)
            if (led_cnt++ > 250)
            {
                HAL_IWDG_Refresh(&hiwdg);     // ~250ms 刷新看门狗
                led_cnt = 0;
                HAL_GPIO_TogglePin(LED1_GPIO_Port, LED1_Pin);  // LED 闪烁指示运行
            }
        }
    }
}
```

### 判决逻辑要点

- 三个阶段的优先级: **RAM 标志 > CAN 请求 > Flash 有效标志**
- `check_app_set_flag()` 读取地址 `0x20000FFC`，值为 `0xABCD1234` 时返回 1
- `can_start_check()` 在延时循环中检测是否收到 `0x10 0x02` (编程会话请求)
- 看门狗在主循环初始化，**判决完成之前不启动**，避免擦写 Flash 时超时复位

## 2. Boot Jump 实现

**文件**: `Core/Src/boot_jump.c` | `Core/Inc/boot_jump.h`

### 关键宏定义

```c
#define APPLICATION_ADDRESS     0x0800C000    // APP 起始地址
#define BACK_TO_BOOT_FLAG_ADDR  (0x20000FFC)  // RAM 回退标志地址
#define BACK_TO_BOOT_FLAG       (0xABCD1234)  // 回退标志值

typedef void (*pFunction)(void);
```

### Boot_JumpToApplication()

```c
void Boot_JumpToApplication(void)
{
    pFunction Jump_To_Application;
    uint32_t JumpAddress;

    // 校验: APP 向量表首字 (MSP) 地址必须在 RAM 范围 0x20000000 内
    if (((*(__IO uint32_t*)APPLICATION_ADDRESS) & 0x2FFE0000) == 0x20000000)
    {
        // 1. 关闭所有外设中断
        for (int i = 0; i < 8; i++) {
            NVIC->ICER[i] = 0xFFFFFFFF;      // 清除中断使能
            NVIC->ICPR[i] = 0xFFFFFFFF;      // 清除中断挂起
        }

        // 2. 全局关中断
        __disable_irq();

        // 3. 复位 RCC 时钟配置
        HAL_RCC_DeInit();

        // 4. 关闭 SysTick
        SysTick->CTRL = 0;                    // 禁止 SysTick
        SysTick->LOAD = 0;                    // 清除重装载值
        SysTick->VAL  = 0;                    // 清除当前值

        // 5. 获取 APP 复位向量地址 (向量表偏移 +4)
        JumpAddress = *(__IO uint32_t*)(APPLICATION_ADDRESS + 4);
        Jump_To_Application = (pFunction)JumpAddress;

        // 6. 设置 MSP 为 APP 的栈顶指针
        __set_MSP(*(__IO uint32_t*)APPLICATION_ADDRESS);

        // 7. 跳转 (不再返回)
        Jump_To_Application();
    }
}
```

### 实现关键点

1. **向量表校验**: 跳转前检查 APP 向量表第一个字（栈顶地址 MSP）是否指向 RAM 区域。这是防止跳转到无效地址的最后一道防线
2. **中断清理**: 必须遍历 8 个 ICER 寄存器清除所有中断使能，再清 ICPR 清除挂起位。遗漏任何一个中断都可能导致 APP 启动后产生意外中断
3. **RCC 复位**: `HAL_RCC_DeInit()` 复位时钟配置到默认状态，让 APP 重新初始化自己的时钟
4. **SysTick 清零**: 如果不清零，APP 可能会在 SysTick 中断处理中访问 FBL 的中断处理函数
5. **MSP 切换**: `__set_MSP()` 必须在 `__disable_irq()` 之后，在跳转之前完成

### Flash 扇区擦除

```c
HAL_StatusTypeDef flash_erase_region(uint32_t startAddr, uint32_t length)
{
    // 参数校验
    if (length == 0u) return HAL_ERROR;
    if (startAddr + length - 1u < startAddr) return HAL_ERROR;  // 整数溢出检查

    uint32_t endAddr = startAddr + length - 1u;
    // 边界检查
    if ((startAddr < FLASH_BASE) || (endAddr < FLASH_BASE) ||
        (startAddr > FLASH_BANK1_END) || (endAddr > FLASH_BANK1_END) ||
        (endAddr < startAddr))
        return HAL_ERROR;

    // 计算起始页和页数 (STM32F103 每页 2KB)
    uint32_t startPageAddr = startAddr & ~(FLASH_PAGE_SIZE - 1u);
    uint32_t nbPages = ((endAddr - startPageAddr) / FLASH_PAGE_SIZE) + 1u;

    FLASH_EraseInitTypeDef eraseInit;
    uint32_t pageError = 0xFFFFFFFFu;
    eraseInit.TypeErase = FLASH_TYPEERASE_PAGES;
    eraseInit.Banks = FLASH_BANK_1;

    if (HAL_FLASH_Unlock() != HAL_OK) return HAL_ERROR;

    HAL_StatusTypeDef flash_ret = HAL_OK;
    for (uint32_t p = 0; p < nbPages; ++p)
    {
        eraseInit.NbPages = 1u;
        eraseInit.PageAddress = startPageAddr + p * FLASH_PAGE_SIZE;
        __disable_irq();
        flash_ret = HAL_FLASHEx_Erase(&eraseInit, &pageError);
        __enable_irq();
        HAL_IWDG_Refresh(&hiwdg);            // 逐页刷新看门狗
        if (flash_ret != HAL_OK) break;
    }
    HAL_FLASH_Lock();
    return flash_ret;
}
```

**实现要点**:
- 逐页擦除，**每页后刷新 IWDG**，防止大区域擦除超时复位（每页擦除耗时约 20-40ms）
- 擦除期间 **关中断**，防止中断干扰 Flash 操作
- 边界检查防止擦除到 Bootloader 自身区域

### Flash 数据写入

```c
HAL_StatusTypeDef flash_write_data(uint32_t addr, const uint8_t* data, uint16_t len)
{
    HAL_StatusTypeDef status = HAL_OK;
    uint32_t word;
    uint8_t bytesToWrite;

    HAL_FLASH_Unlock();

    // 按 4 字节 (字) 写入 — STM32F1 支持的最小编程单位
    for (uint16_t i = 0; i < len; i += 4)
    {
        word = 0;
        bytesToWrite = (len - i < 4) ? (len - i) : 4;
        for (uint8_t j = 0; j < bytesToWrite; j++)
            word |= ((uint32_t)data[i + j] << (j * 8));   // 小端组装

        status = HAL_FLASH_Program(FLASH_TYPEPROGRAM_WORD, addr + i, word);
        if (status != HAL_OK) break;

        if ((i % 8) == 0) HAL_IWDG_Refresh(&hiwdg);       // 每 2 字刷新看门狗
    }
    HAL_FLASH_Lock();
    return status;
}
```

**实现要点**:
- STM32F103 必须按 **字 (32-bit)** 写入，不能逐字节写
- 尾部不足 4 字节时，仅写入有效字节数，高位补 0
- 每写入 2 个字 (8 字节) 刷新看门狗

### Flash 读取

```c
HAL_StatusTypeDef flash_read_4bytes(uint32_t flash_addr, uint8_t *read_buf)
{
    if (read_buf == NULL) return HAL_ERROR;
    if (flash_addr < FLASH_BASE_ADDR || flash_addr + 3 > FLASH_END_ADDR)
        return HAL_ERROR;

    // Flash 读取无需解锁，直接指针读取
    read_buf[0] = *(uint8_t *)(flash_addr + 0);
    read_buf[1] = *(uint8_t *)(flash_addr + 1);
    read_buf[2] = *(uint8_t *)(flash_addr + 2);
    read_buf[3] = *(uint8_t *)(flash_addr + 3);
    return HAL_OK;
}
```

## 4. UDS 服务分发器实现

**文件**: `UDSBase/uds_service.c`

### 服务分发核心函数

```c
static void uds_data_indication(uint8_t* msg_buf, uint16_t msg_dlc, n_result_t n_result)
{
    // 收到完整 UDS 报文时调用，执行 6 步检查
    uds_timer_stop(UDS_TIMER_S3server);     // 先停止 S3Server 定时器

    if (n_result != N_OK) {
        uds_timer_start(UDS_TIMER_S3server);
        return;                             // 传输层错误，不处理
    }

    uint8_t sid = msg_buf[0];
    // 检查 SSP (Suppress Positive Response) 标志 — 子功能字节的 Bit7
    bool_t ssp = (msg_buf[1] & 0x80) ? TRUE : FALSE;

    for (uint8_t i = 0; i < SID_NUM; i++)
    {
        if (sid != uds_service_list[i].uds_sid) continue;

        // ① 功能寻址支持检查
        if (N_TATYPE_FUNCTIONAL == g_tatype && FALSE == uds_service_list[i].fun_spt)
            { uds_no_response(); return; }

        // ② 当前会话支持检查 (标准/编程/扩展)
        if ((UDS_SESSION_STD == get_current_session() && FALSE == uds_service_list[i].std_spt)
            || (UDS_SESSION_PROG == get_current_session() && FALSE == uds_service_list[i].prog_spt)
            || (UDS_SESSION_EXT == get_current_session() && FALSE == uds_service_list[i].ext_spt))
            { uds_negative_rsp(sid, NRC_SERVICE_NOT_SUPPORTED_IN_ACTIVE_SESSION); return; }

        // ③ 安全等级检查
        if (get_current_sa_lv() < uds_service_list[i].uds_sa)
            { uds_negative_rsp(sid, NRC_SECURITY_ACCESS_DENIED); return; }

        // ④ 报文长度检查
        if (FALSE == uds_service_list[i].check_len(msg_buf, msg_dlc))
            { uds_negative_rsp(sid, NRC_INVALID_MESSAGE_LENGTH_OR_FORMAT); return; }

        // ⑤ Suppress Positive Response 处理
        if (uds_service_list[i].ssp_spt == TRUE && ssp == 0x01)
            ssp_flg = TRUE;
        else
            ssp_flg = FALSE;

        // ⑥ 执行服务处理函数
        uds_service_list[i].uds_service(msg_buf, msg_dlc);
        return;
    }

    // 没有匹配的 SID
    uds_negative_rsp(sid, NRC_SERVICE_NOT_SUPPORTED);
}
```

### S3Server 定时器管理

```c
void service_task(void)     // 每 1ms 调用一次
{
    // S3Server 超时 → 退回默认会话 + 清除安全等级
    if (uds_timer_run(UDS_TIMER_S3server) < 0)
    {
        set_current_session(UDS_SESSION_STD);
        set_current_sa_lv(UDS_SA_NON);
    }

    // FSA 超时 → 清除安全访问失败计数
    if (uds_timer_run(UDS_TIMER_FSA) < 0)
    {
        uds_fsa_cnt = 0;
    }
}
```

### 定时器实现

```c
static uint32_t uds_timer[UDS_TIMER_CNT];    // 递减计数器

void uds_timer_start(uint8_t id)
{
    uds_timer[id] = UDS_TIMEOUT_S3SERVER_MS; // 例如: 5000 (5秒)
}

int32_t uds_timer_run(uint8_t id)
{
    if (uds_timer[id] > 0)
        return --uds_timer[id];              // 递减直到 0
    else
        return -1;                            // 超时
}
```

### 接收状态机

```c
void uds_tp_recv_frame(uint8_t func_addr, uint8_t* frame_buf, uint8_t frame_dlc)
{
    if (NULL == frame_buf || FRAME_SIZE != frame_dlc) return;

    g_tatype = (0 == func_addr) ? N_TATYPE_PHYSICAL : N_TATYPE_FUNCTIONAL;
    uint8_t pci_type = NT_GET_PCI_TYPE(frame_buf[0]);

    switch (pci_type)
    {
        case PCI_SF:    // 单帧 — 直接提交给上层
            recv_singleframe(frame_buf, frame_dlc);
            break;

        case PCI_FF:    // 首帧 — 初始化重组缓冲区，回复 FC
            if (NWL_RECV == nwl_st || NWL_IDLE == nwl_st) {
                clear_network();
                int ret = recv_firstframe(frame_buf, frame_dlc);
                nwl_st = (0 == ret) ? NWL_RECV : NWL_IDLE;
            }
            break;

        case PCI_CF:    // 连续帧 — 追加到重组缓冲区
            if (NWL_RECV == nwl_st && TRUE == g_wait_cf) {
                int ret = recv_consecutiveframe(frame_buf, frame_dlc);
                if (ret <= 0) { clear_network(); nwl_st = NWL_IDLE; }
            }
            break;

        case PCI_FC:    // 流控帧 — 继续发送或停止
            if (NWL_XMIT == nwl_st && TRUE == g_wait_fc) {
                int ret = recv_flowcontrolframe(frame_buf, frame_dlc);
                if (ret < 0) { clear_network(); nwl_st = NWL_IDLE; }
            }
            break;
    }
}
```

### 发送接口

```c
{

        send_singleframe(msg_buf, msg_dlc);       // ≤7 字节 → 单帧
```

### 1ms 定时任务

```c
void network_task(void)     // 每 1ms 调用
{
    // N_CR 超时检查 (接收 CF 间隔超时)
    if (N_TIMER_N_CR 超时 && g_wait_cf) { clear_network(); nwl_st = NWL_IDLE; }
    // N_BS 超时检查 (等待 FC 超时)
    if (TIMER_N_BS 超时 && g_wait_fc)  { clear_network(); nwl_st = NWL_IDLE; }
    // STmin 超时 → 发送下一个 CF
    if (STmin 到时 && nwl_st == NWL_XMIT) {
        send_next_cf();
    }
}
```

```c
void service_10_SessionControl(const uint8_t* msg_buf, uint16_t msg_dlc)
```

## 9. 安全访问 (SID 27) 实现

**文件**: `UDSLogic/SID27_SecurityAccess.c`

### 关键宏与状态

```c
#define UDS_SEED_LENGTH       16     // 种子长度 16 字节
#define UDS_KEY_LENGTH        16     // 密钥长度 16 字节
#define UDS_REQUEST_SEED      0x01   // 请求种子子功能
#define UDS_SEND_KEY          0x02   // 发送密钥子功能
#define UDS_FAS_MAX_TIMES     0x02   // 最大失败次数 (2 次)

static uint8_t req_seed = 0;         // 是否已请求种子
static uint8_t org_seed_buf[UDS_SEED_LENGTH];  // 保存已发送的种子
static uds_sa_lv curr_sa = UDS_SA_NON;
uint8_t uds_fsa_cnt = 0;             // 失败计数 (外部引用)
```

### 认证算法 — AES-CMAC

```c
int uds_security_access(uint8_t* key_buf, uint8_t* seed_buf, uint16_t msg_len)
{
    uint8_t AES_CMAC_key[16] = {
        0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff,
        0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff
    };
    uint8_t computed_mac[16];
    AES_CMAC_CTX ctx;

    aes_cmac_init(&ctx, AES_CMAC_key);
    aes_cmac_compute(&ctx, seed_buf, msg_len, computed_mac);

    for (uint8_t i = 0; i < 16; i++) {
        if (computed_mac[i] != key_buf[i])
            return -1;               // 不匹配 → 认证失败
    }
    return 0;                        // 匹配 → 认证通过
}
```

### 种子生成

```c
static void random_seed_generate(uint8_t *seed)
{
    uint32_t cnt = 0;
    srand(HAL_GetTick());            // 用当前 Tick 做随机种子
    for (uint8_t i = 0; i < 16; i++)
    {
        cnt += rand() % 3 + 1;       // 加入随机步进
        if (cnt > 16) cnt = cnt % 16;
        seed[i] ^= rand_u8();        // 异或随机值
        cnt = i;
    }
}
```

### 服务处理状态机

```c
void service_27_SecurityAccess(const uint8_t* msg_buf, uint16_t msg_dlc)
{
    uint8_t subfunction = msg_buf[1] & 0x7f;

    switch (subfunction) {
        case UDS_REQUEST_SEED:       // 0x01 — 请求种子
            // 检查 FSA 定时器是否还在运行 (锁定期间)
            // 已解锁 → 返回全 0 种子
            // 未解锁 → 生成随机种子, 保存到 org_seed_buf, 设置 req_seed=1
            // 发送: 0x67 | 0x01 | seed[16]
            break;

        case UDS_SEND_KEY:           // 0x02 — 发送密钥
            // 检查是否已请求种子 (req_seed == 1)
            // 调用 uds_security_access() 比对 CMAC
            // 匹配 → 设置 SA_LV1, 清空 req_seed, 复位 FSA 计数
            // 不匹配 → uds_fsa_cnt++
            //  若 uds_fsa_cnt >= 2 → 启动 FSA 定时器 (10s 锁定)
            break;

        default:
            uds_negative_rsp(SID_27, NRC_SUBFUNCTION_NOT_SUPPORTED);
    }
}
```

### 安全锁定机制

- **连续失败 2 次** (`UDS_FAS_MAX_TIMES`) 后启动 FSA 定时器，10 秒内拒绝任何安全访问请求
- FSA 超时后自动清零失败计数，恢复可尝试状态
- 切换会话也会清零失败计数

## 10. 例行程序控制 (SID 31) 实现

**文件**: `UDSLogic/SID31_RoutineControl.c`

### 支持的子功能与 RID

| 子功能 | 值 | 说明 |
|--------|-----|------|
| Start | 0x01 | 启动例行程序 |
| Stop | 0x02 | 停止例行程序 (当前不支持) |
| RequestResult | 0x03 | 请求结果 (当前不支持) |

| RID | 功能 | 说明 |
|-----|------|------|
| **0xFF00** | Erase Memory | 擦除指定 Flash 区域 |
| **0x0202** | CRC Check | CRC16-CCITT 完整性校验 |
| **0x0205** | Check Programming Complete | 校验版本并设置 APP 有效标志 |

### 擦除例行程序

```c
void service_31_RoutineControl(const uint8_t* msg_buf, uint16_t msg_dlc)
{
    uint8_t subfunction = msg_buf[1];
    uint16_t rid = ((uint16_t)msg_buf[2] << 8) | msg_buf[3];

    if (UDS_ROUTINE_CTRL_START == subfunction)
    {
        if (rid == 0xFF00u)
        {
            // 报文格式: SID | SF | RID[2] | startAddr[4] | length[4]
            // 从 msg_buf[4..7] 解析起始地址
            // 从 msg_buf[8..11] 解析长度
            uint32_t startAddr = (msg_buf[4] << 24) | (msg_buf[5] << 16) |
                                 (msg_buf[6] << 8)  |  msg_buf[7];
            uint32_t length    = (msg_buf[8] << 24) | (msg_buf[9] << 16) |
                                 (msg_buf[10] << 8) |  msg_buf[11];

            if (HAL_OK == flash_erase_region(startAddr, length))
                uds_positive_rsp(rsp_buf, 4);   // 0x71 | 0x01 | 0xFF | 0x00
            else
                uds_negative_rsp(SID_31, NRC_GENERAL_PROGRAMMING_FAILURE);
        }
        else if (rid == 0x0202u)
        {
            // CRC 校验: 比较 msg_buf[4..5] 与 CRCValue
        }
        else if (rid == 0x0205u)
        {
            // 版本一致性检查: FBL 版本号 == APP 版本号
            // 如果版本匹配且 CRC 通过 → Set_APP_valid_flag()
        }
    }
    else
    {
        // Stop / RequestResult 当前返回 NRC_REQUEST_SEQUENCE_ERROR
        uds_negative_rsp(SID_31, NRC_REQUEST_SEQUENCE_ERROR);
    }
}
```

### 刷写完成流程

完成固件传输后，Tester 调用 `0x31 01 0205`:
1. 比较 FBL 和 APP 的版本号 (从 Flash 固定地址读取)
2. 检查 CRC 校验是否已通过 (`CRCValue` 状态)
3. 全部通过 → 写入 `0x5A5A5A5A` 到 `APP_VALID_FLAG_ADDR`
4. 标记 APP 有效 → 下次复位后跳转到 APP

## 11. 数据下载 (SID 34/36/37) 实现

**文件**: `UDSLogic/SID34_36_37_UpDownloadData.c`

### 全局状态变量

```c
#define DOWNLOAD    1
#define UPLOAD      2
#define maxNumberOfBlockLength 1020        // 每个数据块最大长度

uint8_t  UpDownLoadReq = 0;               // 传输方向 (0=无, 1=下载, 2=上传)
uint8_t  encryptingMethod = 0;
uint8_t  compressionMethod = 0;
uint32_t memoryAddress = 0;               // 目标 Flash 地址
uint32_t memorySize = 0;                  // 总传输大小
uint8_t  blockSequenceCounter = 0;        // 块序号计数器
uint32_t writtenBytes = 0;                // 已写入字节数
volatile uint16_t CRCValue = 0xffff;      // 运行 CRC
```

### SID 34 — RequestDownload

```c
void service_34_RequestDownload(const uint8_t* msg_buf, uint16_t msg_dlc)
{
    // 响应: 最大块长度
    rsp_buf[0] = 0x74;                                       // 正响应 SID
    rsp_buf[1] = 0x20;                                       // 地址/长度格式
    rsp_buf[2] = maxNumberOfBlockLength >> 8;                 // 最大块大小 hi
    rsp_buf[3] = maxNumberOfBlockLength & 0xFF;               // 最大块大小 lo

    // 报文格式解析
    compressionMethod = msg_buf[1] >> 4;                      // 高 4 位: 压缩方法
    encryptingMethod  = msg_buf[1] & 0x0f;                    // 低 4 位: 加密方法
    uint8_t mslen = msg_buf[2] >> 4;                          // 内存大小长度
    uint8_t malen = msg_buf[2] & 0x0f;                        // 内存地址长度

    // 从 msg_buf[3..] 解析地址和大小 (大端序)
    // 校验: mslen ≤ 4, malen ≤ 4
    // 校验: 地址在 0x08000000 ~ 0x0807FFFF 范围内

    UpDownLoadReq = DOWNLOAD;
    writtenBytes = 0;
    blockSequenceCounter = 0;
    CRCValue = 0xffff;                                         // 重置 CRC
    crc_init();                                                // 初始化 CRC 计算
    uds_positive_rsp(rsp_buf, 4);
}
```

### SID 36 — TransferData

```c
void service_36_TransferData(const uint8_t* msg_buf, uint16_t msg_dlc)
{
    uint8_t seqCounter = msg_buf[1];

    // 检查状态合法性
    if (UpDownLoadReq != DOWNLOAD && UpDownLoadReq != UPLOAD) {
        uds_negative_rsp(SID_36, NRC_REQUEST_SEQUENCE_ERROR);
        return;
    }

    // 检查序号连续性
    if (seqCounter - blockSequenceCounter > 1) {
        uds_negative_rsp(SID_36, NRC_WRONG_BLOCK_SEQUENCE_COUNTER);
        return;
    }

    if (DOWNLOAD == UpDownLoadReq) {
        // 去重: 如果序号与上次相同，直接返回成功 (不重复写入)
        if (seqCounter == blockSequenceCounter) {
            // 重复块，跳过写入
        } else {
            uint16_t dataLen = msg_dlc - 2;                     // 减去 SID + 序号
            uint32_t writeAddr = memoryAddress + writtenBytes;
            HAL_StatusTypeDef status = flash_write_data(writeAddr, &msg_buf[2], dataLen);

            if (HAL_OK == status) {
                CRCValue = crc16_ccitt(&msg_buf[2], dataLen);   // 更新运行 CRC
                writtenBytes += dataLen;
            } else {
                uds_negative_rsp(SID_36, NRC_GENERAL_PROGRAMMING_FAILURE);
                return;
            }
        }
        blockSequenceCounter = seqCounter;                      // 更新序号
    }
    uds_positive_rsp(rsp_buf, 2);                               // 0x76 | seqCounter
}
```

### SID 37 — RequestTransferExit

```c
void service_37_RequestTransferExit(const uint8_t* msg_buf, uint16_t msg_dlc)
{
    if (UpDownLoadReq == 0) {
        uds_negative_rsp(SID_37, NRC_REQUEST_SEQUENCE_ERROR);
        return;
    }

    UpDownLoadReq = 0;                                           // 清零传输状态
    uds_positive_rsp(rsp_buf, 1);                                // 0x77
}
```

### 传输协议总结

```
RequestDownload (34)
  │ 协商: Tester 告诉 ECU 要写多少字节
  │ ECU 回复: 每个块最多 1020 字节
  ▼
TransferData (36) × N
  │ 每次携带: 序号 | 数据
  │ ECU 做:   Flash 写入 | 运行 CRC | 累计字节数
  ▼
RequestTransferExit (37)
  │ 结束传输，清理状态
  ▼
RoutineControl (31) — 0x0205
  │ 校验完整性，设置 APP 有效标志
  ▼
ECUReset (11) → 复位 → 跳转 APP
```

## 12. AES-CMAC 实现

### 实现概述

纯软件 AES-128 + CMAC 实现，不依赖硬件加密引擎。所有运算在 Cortex-M3 上以查表和异或方式完成。

### AES 内部函数

```c
static const uint8_t S[256] = {

uint8_t rcon[10] = {0x01, 0x02, 0x04, 0x08, 0x10, 0x20, 0x40, 0x80, 0x1b, 0x36};

// 列混合乘法表 (256 字节)
static const uint8_t gmul2[256] = { ... };    // GF(2^8) ×2 查找表

void add_round_key(uint8_t state[16], const uint8_t rk[16]);   // XOR 轮密钥
void sub_bytes(uint8_t state[16]);                              // S 盒替换
void shift_rows(uint8_t state[16]);                             // 行移位
void mix_columns(uint8_t state[16]);                            // 列混合
void aes_key_expansion(const uint8_t key[16], uint8_t rk[176]); // 密钥扩展 → 11 轮密钥
void aes128_ecb_encrypt(const uint8_t input[16], uint8_t output[16], const uint8_t rk[176]); // 10 轮加密
```

### CMAC 函数

```c
typedef struct {
    uint8_t key[AES_BLOCK_SIZE];    // 128-bit key
} AES_CMAC_CTX;

void generate_subkeys(const uint8_t key[AES_BLOCK_SIZE],
    uint8_t K1[AES_BLOCK_SIZE], uint8_t K2[AES_BLOCK_SIZE])
{
    // 1. 用密钥加密全零块
    // 2. 左移 1 位 → K1 (如果 MSB=1 则异或 0x87)
    // 3. K1 左移 1 位 → K2 (如果 MSB=1 则异或 0x87)
}

void aes_cmac_init(AES_CMAC_CTX* ctx, const uint8_t* key)
{
    memcpy(ctx->key, key, AES_BLOCK_SIZE);
}

void aes_cmac_compute(AES_CMAC_CTX* ctx, const uint8_t* msg,
    uint32_t msg_len, uint8_t out_mac[AES_BLOCK_SIZE])
{
    // NIST SP 800-38B:
    // 1. 生成子密钥 K1, K2
    // 2. 如果最后一块完整 → 异或 K1
    // 3. 如果最后一块不完整 → 补 0x80 + 0x00... 后异或 K2
    // 4. CBC-MAC 计算
}
```

### 内存占用

- 预计算的查找表: S-box (256B) + gmul2 (256B) + rcon (10B) = **约 522B ROM**
- 运行时栈: 轮密钥 (176B) + 状态 (16B) ≈ **200B RAM**
- 适合资源受限的 Cortex-M3 环境

```c
static MsgQueue_S txQueue;          // 发送队列 (主循环 → 中断)

int PutToRxBuf(const Msg_S *msg) {
    // 用 RING_PUT 宏将消息放入 rxQueue
    // 队列满时覆盖最旧的消息
}

int GetFromRxBuf(Msg_S *msg) {
    // 用 RING_GET 宏从 rxQueue 取出消息
    // 队列空时返回 -1
}
```

### 使用模式

```
CAN RX 中断                    主循环                    CAN TX 中断
    │                           │                          │
    ▼                           ▼                          ▲
PutToRxBuf() ──► rxQueue ──► GetFromRxBuf()               │
                    (8 个槽位)       │                     │
                                      ├── uds_tp_recv_frame()
                                      │                     │
                                      └── uds_send_frame()
                                               │            │
                                               ▼            │
                                          PutToTxBuf() ──► txQueue ──► GetFromTxBuf()
```

缓冲区大小 `RING_BUF_SIZE = 8`，队列满时**覆盖最旧消息**，适合 CAN 总线的高速数据流场景。

## 14. 链接脚本与内存布局

**文件**: `STM32F103ZETx_FLASH.ld`

### 内存定义

```
_estack = ORIGIN(RAM) + LENGTH(RAM);        // 0x20000000 + 64K = 0x20010000
_Min_Heap_Size  = 0x200;                     // 最小堆 512 字节
_Min_Stack_Size = 0x400;                     // 最小栈 1KB

MEMORY
{
    RAM    (xrw) : ORIGIN = 0x20000000, LENGTH = 64K
    FLASH  (rx)  : ORIGIN = 0x08000000, LENGTH = 512K
}
```

### 段布局

```
SECTIONS
{
    .isr_vector      : > FLASH              // 中断向量表 @ 0x08000000
    .text            : > FLASH              // 代码段
    .rodata          : > FLASH              // 只读数据
    .data            : > RAM AT> FLASH      // 初始化数据 (RAM 运行, Flash 存储)
    .bss             : > RAM                // 零初始化数据

    .fbl_version 0x08007FFC :               // FBL 版本号固定在 0x08007FFC
    {
        KEEP(*(.fbl_version))
    } > FLASH

    /DISCARD/ : { libc.a, libm.a, libgcc.a }  // 丢弃不需要的标准库
}
```

### 物理内存映射

```
地址范围                   大小   内容
0x08000000 - 0x08007FFF  32KB   FBL 代码 (.isr_vector + .text + .rodata)
0x08007FFC - 0x08007FFF  4B     FBL 版本号 (.fbl_version)
0x08008000 - 0x08009FFF  8KB    隔离区 (未使用)
0x0800A000 - 0x0800A003  4B     APP 有效标志
0x0800C000 - 0x0801FFFF  80KB   APP 代码
0x0801FFFC - 0x0801FFFF  4B     APP 版本号

0x20000000 - 0x2000FFFF  64KB   RAM (.data + .bss + heap + stack)
0x20000FFC - 0x20000FFF  4B     RAM 回退标志 (BACK_TO_BOOT_FLAG)
```

### 链接脚本关键点

- **FBL 版本号** 使用 `__attribute__((section(".fbl_version")))` 固定到地址 `0x08007FFC`，确保位于 FBL 区域末尾
- **APP 起始地址** 为 `0x0800C000`，对应 APPLICATION_ADDRESS 宏
- `.fbl_version` 使用 `KEEP()` 防止链接器优化掉
- 使用 `/DISCARD/` 排除不必要的标准库以减小固件体积

## 15. 看门狗 (IWDG) 实现

**文件**: `Core/Src/iwdg.c` | `Core/Inc/iwdg.h`

### 初始化

```c
void MX_IWDG_Init(void)
{
    hiwdg.Instance = IWDG;
    hiwdg.Init.Prescaler = IWDG_PRESCALER_8;      // 40kHz / 8 = 5kHz
    hiwdg.Init.Reload = 2500;                      // 2500 ticks
    if (HAL_IWDG_Init(&hiwdg) != HAL_OK)
    {
        Error_Handler();
    }
}
```

### 时序计算

```
LSI 时钟:        40 kHz
预分频:           /8 → 5 kHz
Tick 周期:       1/5000 = 200 µs
超时时间:        2500 × 200 µs = 500 ms
刷新周期:        主循环每 ~250ms 刷新一次 (HAL_GetTick() 计数)
```

### 刷新策略

| 位置 | 间隔 | 目的 |
|------|------|------|
| 主循环 `while(1)` | ~250ms | 正常运行保活 |
| 页擦除 `flash_erase_region()` | 每擦一页 | 擦除耗时 ~40ms/页，防止长时间擦除超时 |
| Flash 写入 `flash_write_data()` | 每写 2 字 (8B) | 写入耗时 ~20µs/字，密集写入保活 |
